# ChoralRoot: MIDI start / continue / stop in drive the loop as the LOOP (PLAY) button does, with Options > MIDI Clock
# In and (3y253) Out; Off ignores them. A 1-bar Free loop (as cr_loop_free), LOOP stops it, then In: FC / FA / FC /
# FB / FA-while-playing / FC; Out: FA / FC / FB / FC (none sent back out), then LOOP twice (sent: FA, FC); Off: FA.
keys classic                  # (0.25.1: FM-1 is the default; this script was written for the Classic chord keys)
wait 300
btn GLO 80                     # Options: MIDI Clock -> In
wait 100
knob SELECT +3
btn OCT+ 30
knob SELECT +6
knob KNOB1 +2
wait 70
btn GLO 80
wait 200
btn REC
wait 300
btn A#3 down
btn D4 down
wait 500
btn D4 up
btn A#3 up
wait 500
btn G#3 down
btn E4 down
wait 500
btn E4 up
btn G#3 up
wait 400
btn REC                      # commit: plays
wait 400
expect led GREEN on          # 1
btn PLAY                     # LOOP: stop
wait 300
expect led GREEN off         # 2
midi FC                      # stop while stopped: nothing
wait 200
expect led GREEN off         # 3
midi FA                      # start: plays
wait 200
expect led GREEN on          # 4
dump
midi FC                      # stop
wait 200
expect led GREEN off         # 5
dump
midi FB                      # continue: plays
wait 200
expect led GREEN on          # 6
midi FA                      # start while playing: from the top, still playing
wait 200
expect led GREEN on          # 7
midi FC
wait 200
expect led GREEN off         # 8
btn GLO 80                     # Options: MIDI Clock -> Out
wait 100
knob KNOB1 -1
wait 100
btn GLO 80
wait 200
midi FA                      # Out: plays, as LOOP; no FA sent back
wait 300
expect led GREEN on          # 9
midi FC                      # stops; no FC sent back
wait 300
expect led GREEN off         # 10
midi FB                      # continue: plays
wait 300
expect led GREEN on          # 11
midi FC
wait 300
expect led GREEN off         # 12
wait 1200                    # (past the no-echo window)
btn PLAY                     # the button itself: plays, FA sent
wait 300
expect led GREEN on          # 13
btn PLAY                     # stops, FC sent
wait 300
expect led GREEN off         # 14
btn GLO 80                     # Options: MIDI Clock -> Off
wait 100
knob KNOB1 -1
wait 100
btn GLO 80
wait 200
midi FA                      # Off: ignored
wait 300
expect led GREEN off         # 15
dump
wait 2000
dump
