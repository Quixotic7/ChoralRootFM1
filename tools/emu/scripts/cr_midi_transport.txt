# ignored. A 1-bar Free loop (as cr_loop_free), LOOP stops it, then FC / FA / FC / FB / FA-while-playing; Out: FA.
wait 300
key N 80                     # Options: MIDI Clock -> In
wait 100
knob SELECT +11
knob KNOB1 +2
wait 100
key N 80
wait 200
btn REC
wait 300
btn A#3 down
btn D4 down
wait 500
btn D4 up
btn A#3 up
wait 500
btn G#3 down
btn E4 down
wait 500
btn E4 up
btn G#3 up

wait 400

btn PLAY                     # LOOP: stop
wait 300

wait 200
