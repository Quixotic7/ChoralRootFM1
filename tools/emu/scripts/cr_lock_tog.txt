# ChoralRoot: LOCK with "Ext keys: Toggle" (the default) in the three play styles, the same check as cr_lock_push.txt:
# the latch is the same in every style, what sounds follows it (Simple: the next root; Advanced / Free: the roots held
# too; Free: the last root again while a chord key is held). test_cr.sh runs this body three times behind a preamble
# (Options > Play Style, LOCK on); the last part sets Options > Extension Addition to Play Chord.
#   chord block: F3 6, F#3 DIM, G3 m7, G#3 MIN, A3 M7, A#3 MAJ, B3 LOCK, C4 9, C#4 SUS
#   latched = bits DIM 01 MIN 02 MAJ 04 SUS 08, 6 10, m7 20, M7 40, 9 80
btn A#3                       # MAJ
wait 150
btn D4 200                    # D: the last chord
wait 400
btn F3                        # 6 on
wait 300
btn G3                        # 7 on (Toggle: beside the 6)
wait 300
dump                          # 1: 34, silent
btn G3 down                   # 7 again: off at the press (Free: D6 sounds while it is held)
wait 200
dump                          # 2: 14
btn G3 up
wait 300
dump                          # 3: 14, silent
btn D4 down                   # a root held: D6
wait 300
dump                          # 4: D6
btn F3                        # 6 off: Advanced / Free: D
wait 300
dump                          # 5: 04
btn C4                        # 9 on: D with the 9
wait 300
dump                          # 6: 84
btn C4                        # 9 off again (one push, not two)
wait 300
dump                          # 7: 04
btn C4                        # .. and on
wait 300
dump                          # 8: 84
btn A#3                       # the top row's lone repeat: no type: Advanced / Free: a single note, the 9 waiting
wait 300
dump                          # 9: 80
btn G#3                       # MIN: the type back (Clears: no type was latched, so the 9 stays): Dm with the 9
wait 300
dump                          # 10: 82
btn A#3                       # MAJ alone (Clears): the extensions go: D
wait 300
dump                          # 11: 04
btn D4 up
wait 300
dump                          # 12: silent
# Options > Extension Addition: Play Chord (SELECT +1 from Play Style, KNOB 1 +1): a change re-strikes the chord,
# once per push
btn GLO
wait 200
btn OCT+ 30
knob SELECT +1
wait 170
knob KNOB1 +1
wait 200
btn GLO
wait 300
btn D4 down
wait 300
dump                          # 13: D
btn C4                        # 9 on
wait 300
dump                          # 14: 84
btn C4                        # 9 off
wait 300
dump                          # 15: 04
btn D4 up
wait 300
btn B3                        # LOCK off
wait 2500
dump                          # 16: 00, silent
