# ChoralRoot: Key Mode's lights on the root keys (the owner, 2026-10-11 / 12): four levels: faint < dim < mid < lit.
# Key Mode off: the roots are dark (no idle glow) unless they sound; the chord keys and B3 idle at the faint level
# (under the glow), with Key Mode on too. Key Mode on, everywhere the roots play: the notes of the scale glow (dim, fm1_led_dim), the tonic's
# keys at the middle level (mid, fm1_led_dim2), the roots out of the scale are dark, a sounding note is lit over it
# (on every root of its name: both D of Dm, the tonic's mid under them).
# The dump's "key leds:" line: F3..C4, then D4..G5: 0 off 1 faint 2 dim 3 mid 4 lit.
keys classic                  # (0.25.1: FM-1 is the default; this script was written for the Classic chord keys)
wait 300
expect led D4 off             # Key Mode off: a root has no idle glow,
expect led G5 off
expect led F3 faint           # a chord key idles faint (DIM .. 9, B3 = LOCK),
expect led C4 faint
expect led B3 faint
expect led FX on              # the buttons as ever: FX on,
expect led EDIT dim           # an idle one glows (dim: over the chord keys)
dump                          # 1: Key Mode off: 111111111 000000000000000000
shot build/emu/test/cr_key_leds_off.ppm
btn SEL                       # tap: Key Mode on, C major
wait 300
expect led C5 mid             # the tonic
expect led D4 dim             # a note of the scale
expect led D#4 off            # out of it
expect led F#4 off
expect led B4 dim             # (a white key in C major)
dump                          # 2: 202202020230202202 (the chord keys still 1)
shot build/emu/test/cr_key_leds_c.ppm
btn SEL down                  # the layer: the tonic D (KNOB 1), still major
wait 500
btn SEL up
wait 300
knob KNOB1 +2
wait 300
dump                          # 3: in the layer (keys: pick): the same picture: 302022020202302022
btn OCT-
wait 300
expect led D4 mid
expect led D5 mid
expect led F4 off             # a white key out of D major: dark (it glowed until 0.17)
expect led C5 off
expect led F#4 dim
expect led C#5 dim
dump                          # 4: D major: 302022020202302022
btn SEL down                  # D minor (KNOB 2)
wait 500
btn SEL up
wait 300
knob KNOB2 +1
wait 300
btn OCT-
wait 300
dump                          # 5: D minor: 302202022020302202
btn D4 down                   # a lone root: Dm (D4 F4 A4) lit over the glow
wait 400
expect led D4 on
expect led F4 on
expect led A4 on
expect led E4 dim
expect led D5 on              # (the other D: lit too, over the tonic's mid)
expect led F#4 off
dump                          # 6: 402402042020402402
shot build/emu/test/cr_key_leds_dm.ppm
btn D4 up
wait 300
btn SEL                       # Key Mode off: the roots dark again
wait 300
expect led D#4 off
expect led D4 off
dump                          # 7: 000000000000000000
btn GLO 80                    # Options > LEDs: Stock (two levels: the idle keys lit, the active dark)
wait 300
knob SELECT +24
wait 300
knob KNOB1 +1
wait 300
btn GLO 80
wait 300
btn SEL                       # Key Mode on (D minor): the scale's keys lit, the tonic as the rest, the others dark
wait 300
expect led D4 on
expect led E4 on
expect led D#4 off
expect led F#4 off
dump                          # 8: Stock: 404404044040404404, the chord keys 444444444
btn D4 down                   # Stock: the sounding notes are the dark ones (the reverse, as ever)
wait 400
expect led D4 off
expect led D5 off             # (every root of a sounding note's name: the same keys as Glow lights)
expect led F5 off
expect led E4 on
btn D4 up
wait 2500
dump                          # 9: silent
btn GLO 80                    # LEDs: Glow again; Key Mode off: a held root is the only root light
wait 300
knob KNOB1 -1
wait 300
btn GLO 80
wait 300
btn SEL
wait 300
btn D4 down
wait 400
expect led D4 on
expect led E4 off
dump                          # 10: 400000000000400000 (the single note D: both D)
btn D4 up
wait 300
btn GLO 80                    # Stock, Key Mode off: as it was: every root and chord key lit (idle = lit)
wait 300
knob KNOB1 +1
wait 300
btn GLO 80
wait 300
expect led D4 on
expect led F3 on
expect led B3 on
wait 2500
dump                          # 11: Stock: 444444444 444444444444444444
