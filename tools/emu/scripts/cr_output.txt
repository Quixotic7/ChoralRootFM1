# ChoralRoot 0.19: Options > Output: Classic / Clean (docs/AUDIO-QUALITY.md), switched while a chord sounds.
# tools/emu/test_cr.sh: the three dumps' "master:" lines (Clean, the chain 1; Classic, 0; Clean again), the value on
# the screen, and the WAV: no jump, no hole, no click at either change (tools/emu/wavclicks.py).
# A 4-note chord on TINE EP held throughout; OPT tap, SELECT to Output (entry 21, after Dual Core); KNOB 1 -1: Classic;
# 1 s later +1: Clean; OPT closes; the chord let go.
keys classic                  # (0.25.1: FM-1 is the default; this script was written for the Classic chord keys)
wait 500
key F3 down
key 2 down
key 4 down
key 5 down
wait 40
btn D4 down
wait 1500
dump
btn GLO 80
wait 300
knob SELECT +5
btn OCT+ 30
wait 270
shot build/emu/test/cr_output_clean.ppm
knob KNOB1 -1
wait 600
dump
shot build/emu/test/cr_output_classic.ppm
wait 600
knob KNOB1 +1
wait 600
dump
btn GLO 80
wait 600
btn D4 up
key F3 up
key 2 up
key 4 up
key 5 up
wait 2500
dump
quit
