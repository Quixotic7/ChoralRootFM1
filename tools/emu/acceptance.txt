# Headless check of the harness on Felucca's UI:  build/host/emu --headless --script tools/emu/acceptance.txt

wait 500ms
shot acc_home                # HOME
press A 300ms                # D4 sounds, its LED lit
press C                      # ENV page

shot acc_env

press UP                     # next preset
wait 400ms
shot acc_preset
dump

wait 300ms
shot acc_seq
press ,

press BKSP                   # REC: arms track 1, starts PLAY
wait 200ms
press A 200ms
wait 100ms
press D 200ms
wait 100ms
press G 200ms
wait 600ms
press BKSP                   # disarm, keeps playing
wait 800ms
shot acc_playback
dump                         # T1: 3 steps with notes, playing 1
